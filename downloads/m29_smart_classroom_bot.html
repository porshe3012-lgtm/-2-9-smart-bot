1. การเตรียมฐานข้อมูล (database_setup.py)

รันไฟล์นี้เพียงครั้งเดียวเพื่อสร้างตารางที่จำเป็นทั้งหมด

import sqlite3

def init_db():
    conn = sqlite3.connect('classroom.db')
    cursor = conn.cursor()

    # ตารางการบ้าน
    cursor.execute('''
    CREATE TABLE IF NOT EXISTS homework (
        id INTEGER PRIMARY KEY AUTOINCREMENT,
        subject TEXT,
        details TEXT,
        deadline TEXT,
        status TEXT DEFAULT 'pending'
    )''')

    # ตารางเวรยกหนังสือ
    cursor.execute('''
    CREATE TABLE IF NOT EXISTS book_duty (
        id INTEGER PRIMARY KEY,
        last_student_index INTEGER
    )''')
    cursor.execute('INSERT OR IGNORE INTO book_duty (id, last_student_index) VALUES (1, 0)')

    # ตารางตารางเรียน (ตัวอย่างสำหรับวิชา ส22101)
    cursor.execute('''
    CREATE TABLE IF NOT EXISTS schedule (
        id INTEGER PRIMARY KEY AUTOINCREMENT,
        day TEXT,
        time TEXT,
        subject_code TEXT
    )''')
    
    # เพิ่มข้อมูลตารางเรียนวิชาสังคม/ประวัติ
    schedule_data = [
        ('จันทร์', '12:10', 'ส22101'),
        ('อังคาร', '13:00', 'ส22101'),
        ('พุธ', '12:10', 'ส22101'),
        ('ศุกร์', '09:40', 'ส22101')
    ]
    cursor.executemany('INSERT INTO schedule (day, time, subject_code) VALUES (?, ?, ?)', schedule_data)

    conn.commit()
    conn.close()
    print("Database Initialized Successfully!")

if __name__ == "__main__":
    init_db()

def send_message(reply_token, text):
    url = 'https://api.line.me/v2/bot/message/reply'
    headers = {
        'Content-Type': 'application/json',
        'Authorization': f'Bearer {ACCESS_TOKEN}'
    }
    data = {
        'replyToken': reply_token,
        'messages': [{'type': 'text', 'text': text}]
    }
    requests.post(url, headers=headers, data=json.dumps(data))

def get_book_duty():
    conn = sqlite3.connect('classroom.db')
    cursor = conn.cursor()
    cursor.execute('SELECT last_student_index FROM book_duty WHERE id = 1')
    last_idx = cursor.fetchone()[0]
    
    # รันเลขที่ถัดไป 2 คน

    s2 = (s1 % 40) + 1

def add_homework(details):
    # ตัวอย่างการรับค่า: "คณิต หน้า 5 ส่งพรุ่งนี้"
    try:
        parts = details.split(' ')
        subject = parts[0]
        work = parts[1]
        conn = sqlite3.connect('classroom.db')
        cursor = conn.cursor()
        cursor.execute('INSERT INTO homework (subject, details) VALUES (?, ?)', (subject, work))
        conn.commit()
        conn.close()
        return f"✅ บันทึกงานวิชา {subject} เรียบร้อยแล้ว!"
    except:
        return "❌ รูปแบบผิด! โปรดพิมพ์: เพิ่มงาน [วิชา] [รายละเอียด]"

def list_homework():
    conn = sqlite3.connect('classroom.db')
    cursor = conn.cursor()
    cursor.execute("SELECT subject, details FROM homework WHERE status = 'pending'")
    rows = cursor.fetchall()
    conn.close()
    if not rows: return "ตอนนี้ไม่มีงานค้างครับ! ✨"
    res = "📌 รายการงานที่ต้องส่ง:\n"
    for r in rows:
        res += f"- {r[0]}: {r[1]}\n"
    return res

@app.route("/callback", methods=['POST'])
def callback():
    body = request.get_data(as_text=True)
    data = json.loads(body)
    
    for event in data['events']:
        if event['type'] == 'message':
            msg_text = event['message']['text']
            reply_token = event['replyToken']
            
            if msg_text == 'ใครยกหนังสือ':
                send_message(reply_token, get_book_duty())
            elif msg_text == 'สุ่มเลขที่':
                send_message(reply_token, random_student())
            elif msg_text == 'ดูการบ้าน':
                send_message(reply_token, list_homework())
            elif msg_text.startswith('เพิ่มงาน'):
                content = msg_text.replace('เพิ่มงาน', '').strip()
                send_message(reply_token, add_homework(content))
            else:
                send_message(reply_token, "สวัสดีครับ! ผมบอท ม.2/9\nลองกดปุ่มเมนูหรือพิมพ์ 'ดูการบ้าน' นะครับ")
                
    return 'OK'
